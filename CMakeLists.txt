cmake_minimum_required(VERSION 3.25)
project(ArcReadyPanel VERSION 0.2.18 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "ArcGlass requires Windows 11 x64")

target_compile_options(ArcGlass PRIVATE /W4 /permissive- /utf-8)
